text = input("Enter a string (max 200 characters): ")

uppercase = 0
lowercase = 0
digits = 0
spaces = 0

i = 0
length = len(text)

while i < length:

    ch = text[i]

    if ch == '.':
        break

    if not (ch.isupper() or ch.islower() or ch.isdigit() or ch.isspace()):
        i += 1
        continue

    if ch.isupper():
        uppercase += 1
    elif ch.islower():
        lowercase += 1
    elif ch.isdigit():
        digits += 1
    elif ch.isspace():
        spaces += 1

    i += 1


print(f"Uppercase letters:", {uppercase})
print(f"Lowercase letters:", {lowercase})
print(f"Digits:", {digits})
print(f"Spaces:", {spaces})
